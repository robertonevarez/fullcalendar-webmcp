The FullCalendar WebMCP integration works seamlessly in Next.js applications, including the Next.js App Router and Pages Router.

## Supported characteristics

- **Client Component:** The [`useFullCalendarWebMCP`](/integrations/fullcalendar/reference/use-fullcalendar-webmcp) hook runs in client components. The package entry bundle automatically includes the `'use client';` directive.
- **SSR Safe:** The integration does not access browser globals (`window`, `document`) at module evaluation time. Next.js server-side rendering (SSR) and static generation (prerendering) execute without throwing errors.
- **Server Actions & API Routes:** Your [`CalendarEventRepository`](/integrations/fullcalendar/reference/calendar-event-repository) implementation can call Next.js Server Actions, Route Handlers (`app/api/...`), or ORM-backed services directly.

## App Router example

Create a client component for your calendar and connect it to your server actions or API:

```tsx app/calendar/calendar-client.tsx
"use client";

import FullCalendar from "@fullcalendar/react";
import dayGridPlugin from "@fullcalendar/daygrid";
import { useCallback, useRef, useState } from "react";
import {
  useFullCalendarWebMCP,
  type CalendarEvent,
  type CalendarEventRepository,
} from "protocoltooling";
import {
  fetchEventsAction,
  createEventAction,
  updateEventAction,
  deleteEventAction,
} from "./actions";

export function CalendarClient({
  initialEvents,
}: {
  initialEvents: CalendarEvent[];
}) {
  const calendarRef = useRef<FullCalendar>(null);
  const [events, setEvents] = useState<CalendarEvent[]>(initialEvents);

  const repository: CalendarEventRepository = {
    async list(query) {
      return fetchEventsAction(query);
    },
    async get(id) {
      const all = await fetchEventsAction();
      return all.find((e) => e.id === id) ?? null;
    },
    async create(input) {
      return createEventAction(input);
    },
    async update(id, input) {
      return updateEventAction(id, input);
    },
    async delete(id) {
      await deleteEventAction(id);
    },
  };

  const refetch = useCallback(async () => {
    const updated = await fetchEventsAction();
    setEvents(updated);
  }, []);

  useFullCalendarWebMCP({
    calendarRef,
    events: repository,
    onEventsChanged: refetch,
  });

  return (
    <FullCalendar
      ref={calendarRef}
      plugins={[dayGridPlugin]}
      initialView="dayGridMonth"
      events={events.map((e) => ({ ...e, end: e.end ?? undefined }))}
    />
  );
}
```

Then render it from a server component page:

```tsx app/calendar/page.tsx
import { CalendarClient } from "./calendar-client";
import { fetchEventsAction } from "./actions";

export default async function CalendarPage() {
  const events = await fetchEventsAction();

  return (
    <main className="p-4">
      <h1 className="text-2xl font-bold mb-4">Team Schedule</h1>
      <CalendarClient initialEvents={events} />
    </main>
  );
}
```
